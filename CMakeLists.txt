cmake_minimum_required(VERSION 3.16)
project(ImageMorph LANGUAGES CXX)
add_executable(ImageMorph WIN32 src/main.cpp)
target_compile_features(ImageMorph PRIVATE cxx_std_17)
target_link_libraries(ImageMorph PRIVATE gdiplus comdlg32 gdi32 ole32)
if(MINGW)
  target_compile_options(ImageMorph PRIVATE -finput-charset=UTF-8)
  target_link_options(ImageMorph PRIVATE -municode -static)
endif()
if(MSVC)
  target_compile_options(ImageMorph PRIVATE /utf-8)
endif()
